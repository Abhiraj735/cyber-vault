function startGame() {

    currentPuzzle = 0;
    code = [];
    timeLeft = 120;

    document.getElementById("startScreen")
        .classList.remove("active");

    document.getElementById("gameScreen")
        .classList.add("active");

    showPuzzle();

    clearInterval(timerInterval);

    timerInterval = setInterval(() => {

        timeLeft--;

        document.getElementById("timer")
            .textContent = timeLeft;

        if (timeLeft <= 0) {

            clearInterval(timerInterval);

            alert("⏰ Time's up!");

            location.reload();
        }

    }, 1000);
}


function showPuzzle() {

    document.getElementById("puzzleNumber")
        .textContent =
        "Challenge " + (currentPuzzle + 1) + " / 5";

    document.getElementById("progressBar")
        .style.width =
        ((currentPuzzle) / 5 * 100) + "%";

    document.getElementById("puzzleArea")
        .innerHTML =
        puzzles[currentPuzzle].question;

    document.getElementById("message")
        .className = "message";

    document.getElementById("message")
        .innerHTML = "";
}


function answer(puzzle, result) {

    const message =
        document.getElementById("message");

    if (result === "correct") {

        code.push(puzzles[puzzle].digit);

        message.className =
            "message correct";

        message.innerHTML =
            "🟢 CORRECT!<br><br>" +
            "Security digit unlocked: <strong>" +
            puzzles[puzzle].digit +
            "</strong>";

        setTimeout(() => {

            currentPuzzle++;

            if (currentPuzzle >= puzzles.length) {

                finishGame();

            } else {

                showPuzzle();

            }

        }, 1500);

    } else {

        message.className =
            "message wrong";

        message.innerHTML =
            "🔴 INCORRECT!<br><br>" +
            "Try again.";
    }
}


function finishGame() {

    clearInterval(timerInterval);

    document.getElementById("gameScreen")
        .classList.remove("active");

    document.getElementById("finalScreen")
        .classList.add("active");

    document.getElementById("codeDisplay")
        .textContent =
        code.join(" ");
}

    const entered =
        document.getElementById("finalCode")
        .value;

    const result =
        document.getElementById("vaultResult");

    if (entered === "73815") {

        result.innerHTML = `
            <div class="vault">
                <div class="vault-icon">🔓</div>

                </p>
            </div>
        `;

    } else {

        result.innerHTML = `
            <div class="message wrong">
                ❌ ACCESS DENIED<br><br>
                Incorrect security code.
            </div>
        `;
    }
}
